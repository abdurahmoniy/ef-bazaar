package uz.efbazaar.payment.plan;

import org.flywaydb.core.Flyway;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.jdbc.core.JdbcTemplate;
import uz.efbazaar.payment.testsupport.AbstractPostgresIT;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * Prices and slot counts belong to the admin after the first deploy.
 *
 * <p>The repeatable migration re-runs whenever its content changes; if it still
 * wrote these values, the next unrelated edit to it would quietly put every
 * admin decision back to the shipped defaults.
 */
@SpringBootTest
class PlanMigrationIT extends AbstractPostgresIT {

    @Autowired JdbcTemplate jdbc;
    @Autowired Flyway flyway;

    @Test
    void theShippedPricesAreSeeded() {
        assertThat(jdbc.queryForObject(
                "SELECT price_tiyin FROM plan_prices WHERE plan_code='PRO' AND period='MONTH'", Long.class))
                .isEqualTo(1_000_000L);
        assertThat(jdbc.queryForObject(
                "SELECT price_tiyin FROM plan_prices WHERE plan_code='MAX' AND period='YEAR'", Long.class))
                .isEqualTo(25_000_000L);
        assertThat(jdbc.queryForObject(
                "SELECT count(*) FROM plan_prices WHERE plan_code='STANDARD'", Long.class))
                .as("the free plan is not for sale").isZero();
    }

    @Test
    void theRepeatableMigrationDoesNotOwnAdminManagedValues() {
        jdbc.update("UPDATE plan_entitlements SET value='7' WHERE plan_code='PRO' AND key='mailbox.slots'");
        jdbc.update("UPDATE plan_prices SET price_tiyin=1234500 WHERE plan_code='PRO' AND period='MONTH'");
        try {
            // Forces every repeatable to run again, as a content change would.
            jdbc.update("DELETE FROM flyway_schema_history WHERE type='SQL' AND version IS NULL");
            flyway.migrate();

            assertThat(jdbc.queryForObject(
                    "SELECT value FROM plan_entitlements WHERE plan_code='PRO' AND key='mailbox.slots'",
                    String.class)).isEqualTo("7");
            assertThat(jdbc.queryForObject(
                    "SELECT price_tiyin FROM plan_prices WHERE plan_code='PRO' AND period='MONTH'",
                    Long.class)).isEqualTo(1_234_500L);
        } finally {
            jdbc.update("UPDATE plan_entitlements SET value='5' WHERE plan_code='PRO' AND key='mailbox.slots'");
            jdbc.update("UPDATE plan_prices SET price_tiyin=1000000 WHERE plan_code='PRO' AND period='MONTH'");
        }
    }
}

package uz.efbazaar.payment.plan.domain;

import java.time.Instant;
import java.time.ZoneOffset;

/** How long one purchase lasts. Calendar months and years, counted in UTC. */
public enum PlanPeriod {
    MONTH, YEAR;

    public Instant addTo(Instant from) {
        var at = from.atZone(ZoneOffset.UTC);
        return (this == MONTH ? at.plusMonths(1) : at.plusYears(1)).toInstant();
    }
}

package uz.efbazaar.payment.plan.domain;

import jakarta.persistence.*;
import lombok.*;

import java.time.Instant;
import java.util.UUID;

/** What one period of a plan costs. Edited by admins; seeded once by V15. */
@Entity
@Table(name = "plan_prices")
@Getter @Setter @NoArgsConstructor @AllArgsConstructor @Builder
public class PlanPrice {
    @EmbeddedId
    private PlanPriceId id;

    @Column(name = "price_tiyin", nullable = false)
    private long priceTiyin;

    @Column(name = "updated_by")
    private UUID updatedBy;

import lombok.RequiredArgsConstructor;

import org.springframework.stereotype.Service;

import uz.efbazaar.payment.plan.repository.PlanRepository;

/**
 * What plans cost. Admins edit it; nobody else writes it.
 *
 * <p>A plan with no price row is not for sale — that is how STANDARD stays free
 * without a special case in the purchase path.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class PlanPriceService {

    private final PlanPriceRepository prices;
    private final PlanRepository plans;

    @Transactional(readOnly = true)
    public OptionalLong priceOf(String planCode, PlanPeriod period) {
        return prices.findById(new PlanPriceId(planCode, period))
                .map(p -> OptionalLong.of(p.getPriceTiyin()))
                .orElse(OptionalLong.empty());
    }

    @Transactional(readOnly = true)
    public Map<PlanPeriod, Long> pricesOf(String planCode) {
        Map<PlanPeriod, Long> out = new EnumMap<>(PlanPeriod.class);
        for (PlanPrice p : prices.findByIdPlanCode(planCode)) {
            out.put(p.getId().getPeriod(), p.getPriceTiyin());
        }
        return out;
    }

    @Transactional(readOnly = true)
    public List<PlanPriceRow> all() {
        return prices.findAllByOrderByIdPlanCodeAscIdPeriodAsc().stream().map(PlanPriceRow::of).toList();
    }

    /**
     * Applies all rows or none. STANDARD is refused: pricing the free plan would
     * make it purchasable, which is not a price change but a product change.
     */
    @Transactional
    public List<PlanPriceRow> update(UUID adminId, List<UpdatePlanPricesRequest.Item> items) {
        for (var item : items) {
            if (!plans.existsById(item.planCode()) || PlanService.DEFAULT_PLAN.equals(item.planCode())) {
                throw new IllegalArgumentException("Not a purchasable plan: " + item.planCode());
            }
            var id = new PlanPriceId(item.planCode(), item.period());
            PlanPrice row = prices.findById(id).orElseGet(() -> PlanPrice.builder().id(id).build());
            row.setPriceTiyin(item.priceTiyin());
            row.setUpdatedBy(adminId);
            prices.save(row);
            log.info("Admin {} set {} {} price to {} tiyin", adminId, item.planCode(), item.period(), item.priceTiyin());
        }
        return all();
    }

package uz.efbazaar.payment.plan.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;
import uz.efbazaar.payment.plan.domain.PlanEntitlement;
import uz.efbazaar.payment.plan.domain.PlanEntitlementId;
import uz.efbazaar.payment.plan.dto.PlanEntitlementRow;
import uz.efbazaar.payment.plan.dto.PlanEntitlementsSnapshot;
import uz.efbazaar.payment.plan.dto.UpdatePlanEntitlementsRequest;
import uz.efbazaar.payment.plan.repository.PlanEntitlementRepository;
import uz.efbazaar.payment.plan.repository.PlanRepository;

import java.time.Instant;
import java.util.*;

/**
 * The entitlements an admin may change from the panel.
 *
 * <p>Only the keys listed here: anything else is either not a number of things
 * or not ready to be tuned by hand, and accepting it would let a typo grant a
 * capability nothing enforces yet.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class PlanEntitlementAdminService {

    public static final Set<String> ADMIN_KEYS = Set.of("mailbox.slots", "mailbox.custom_slots");

    private final PlanEntitlementRepository entitlements;
    private final PlanRepository plans;
    private final PlanService planService;
    private final PlanEventPublisher events;

    @Transactional(readOnly = true)
    public List<PlanEntitlementRow> all() {
        return entitlements.findAll().stream()
                .filter(e -> ADMIN_KEYS.contains(e.key()))
                .sorted(Comparator.comparing(PlanEntitlement::planCode).thenComparing(PlanEntitlement::key))
                .map(PlanEntitlementRow::of)
                .toList();
    }

    @Transactional
    public List<PlanEntitlementRow> update(UUID adminId, List<UpdatePlanEntitlementsRequest.Item> items) {
        items.forEach(PlanEntitlementAdminService::validate);   // all before any write
        Set<String> touched = new TreeSet<>();
        for (var item : items) {
            if (!plans.existsById(item.planCode())) {
                throw new IllegalArgumentException("Unknown plan " + item.planCode());
            }
            var id = new PlanEntitlementId(item.planCode(), item.key());
            PlanEntitlement row = entitlements.findById(id)
                    .orElseGet(() -> PlanEntitlement.builder().id(id).build());
            row.setValue(item.value() == null ? null : item.value().trim());
            row.setUpdatedBy(adminId);
            entitlements.save(row);
            touched.add(item.planCode());
            log.info("Admin {} set {}.{} = {}", adminId, item.planCode(), item.key(),
                    item.value() == null ? "unlimited" : item.value());
        }
        // After commit: a reader must never see a value that was rolled back.
        TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
            @Override public void afterCommit() {
                for (String plan : touched) {
                    events.planEntitlements(new PlanEntitlementsSnapshot(
                            plan, planService.entitlementsOf(plan), Instant.now()));
                }
            }
        });
        return all();
    }

    private static void validate(UpdatePlanEntitlementsRequest.Item item) {
        if (!ADMIN_KEYS.contains(item.key())) {
            throw new IllegalArgumentException("Not an admin-managed key: " + item.key());
        }
        if (item.value() == null) return;
        try {
            if (Long.parseLong(item.value().trim()) < 0) {
                throw new IllegalArgumentException("Negative " + item.key());
            }
        } catch (NumberFormatException e) {
            throw new IllegalArgumentException("Not a whole number: " + item.value());
        }
    }
}

// plan/service/PlanTerm.java
package uz.efbazaar.payment.plan.service;

import org.springframework.http.HttpStatus;
import uz.efbazaar.payment.plan.domain.PlanPeriod;
import uz.efbazaar.payment.plan.exception.PlanPurchaseException;

import java.time.Instant;

/**
 * When a purchased plan starts and ends. Pure, so every rule is a unit test.
 *
 * <p>Same plan extends; a higher one starts today at full length and the rest of
 * the old one is forfeited (product decision); a lower one over a live higher
 * one is refused, because the buyer would pay to lose what they already have.
 */
public final class PlanTerm {

    private PlanTerm() {}

    /** {@code expiresAt == null} = does not run out. A free/expired user is STANDARD with null. */
    public record Current(String planCode, int sortOrder, Instant expiresAt) {}

    public record Window(Instant startsAt, Instant expiresAt) {}

    public static Window next(Current current, String planCode, int targetOrder,
                              PlanPeriod period, Instant now) {
        boolean onFree = PlanService.DEFAULT_PLAN.equals(current.planCode());
        if (onFree) {
            return new Window(now, period.addTo(now));
        }
        if (current.planCode().equals(planCode)) {
            if (current.expiresAt() == null) {
                throw new PlanPurchaseException(HttpStatus.CONFLICT, "already_unlimited",
                        "Bu tarif sizda muddatsiz faol");
            }

/**
 * Buying a plan period from the wallet. One transaction: the debit, the history
 * row and the subscription commit together or not at all.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class PlanPurchaseService {

    private final PlanRepository plans;
    private final PlanPriceService prices;
    private final PlanService planService;
    private final UserSubscriptionRepository subscriptions;
    private final PlanPurchaseRepository purchases;
    private final LedgerService ledger;
    private final Clock clock;

    @Transactional
    public MyPlanResponse purchase(UUID userId, Long telegramId, PurchasePlanRequest req) {
        var replay = purchases.findByIdempotencyKey(req.idempotencyKey());
        if (replay.isPresent()) {
            return mine(userId);
        }

        Plan target = plans.findById(req.planCode()).filter(Plan::isActive)
                .filter(p -> !PlanService.DEFAULT_PLAN.equals(p.getCode()))
                .orElseThrow(() -> new PlanPurchaseException(HttpStatus.BAD_REQUEST,
                        "plan_not_purchasable", "Bu tarifni sotib olib bo'lmaydi"));
        long price = prices.priceOf(target.getCode(), req.period())
                .orElseThrow(() -> new PlanPurchaseException(HttpStatus.BAD_REQUEST,
                        "plan_not_purchasable", "Bu tarifni sotib olib bo'lmaydi"));
        if (price != req.expectedPriceTiyin()) {
            throw new PlanPurchaseException(HttpStatus.CONFLICT, "price_changed",
                    "Tarif narxi o'zgardi", Map.of("currentPriceTiyin", price));
        }

        subscriptions.ensureRow(userId);
        var sub = subscriptions.lockByUserId(userId).orElseThrow();
        Instant now = clock.instant();
        boolean live = sub.isLiveAt(now) && !"SYSTEM".equals(sub.getSource());
        String curCode = live ? sub.getPlanCode() : PlanService.DEFAULT_PLAN;
        int curOrder = plans.findById(curCode).map(Plan::getSortOrder).orElse(0);
        var window = PlanTerm.next(new PlanTerm.Current(curCode, curOrder, live ? sub.getExpiresAt() : null),
                target.getCode(), target.getSortOrder(), req.period(), now);

        long available = ledger.balanceOf(userId).getAvailableTiyin();
        if (available < price) {
            // Word for word the shape the frontend's isInsufficientFunds matches.
            throw new ResponseStatusException(HttpStatus.CONFLICT,
                    "Hamyonda mablag' yetarli emas — " + ((price - available + 99) / 100)
                            + " so'm yetishmayapti");
        }
        UUID tx = ledger.post(TransactionKind.PLAN_PURCHASE, "plan-purchase:" + req.idempotencyKey(), userId,
                "plan " + target.getCode() + " " + req.period(),
                List.of(new Posting(LedgerAccount.USER_AVAILABLE, userId, price),
                        new Posting(LedgerAccount.PLATFORM_REVENUE, null, -price)));

        purchases.save(PlanPurchase.builder()
                .id(UUID.randomUUID()).userId(userId).planCode(target.getCode()).period(req.period())
                .priceTiyin(price).startsAt(window.startsAt()).expiresAt(window.expiresAt())
                .ledgerTxId(tx).idempotencyKey(req.idempotencyKey()).telegramId(telegramId)
                .createdAt(now).build());
        planService.assign(userId, target.getCode(), window.expiresAt(), "PURCHASE");
        log.info("User {} bought {} {} for {} tiyin, until {}", userId, target.getCode(),
                req.period(), price, window.expiresAt());
        return mine(userId);
    }

    private MyPlanResponse mine(UUID userId) {
        String code = planService.planFor(userId);
        return new MyPlanResponse(code, planService.expiryFor(userId), planService.entitlementsOf(code));
    }
}

import lombok.RequiredArgsConstructor;

/**
 * Tells a buyer their plan is about to end, and that it has.
 *
 * <p>Single instance, like InvoiceExpirySweeper. A purchase that a later one
 * superseded is marked reminded without sending: the user is still live and
 * "your plan ends" would be false.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class PlanReminderJob {

    private final PlanPurchaseRepository purchases;
    private final UserSubscriptionRepository subscriptions;
    private final AdminTelegramNotifier notifier;
    private final Clock clock;

    @Scheduled(fixedDelayString = "${payment.scheduler.plan-reminder-interval-ms:3600000}")
    @Transactional

            if (stillTheEndOf(p)) notifier.planReminderToOwner(p.getTelegramId(), p.getPlanCode(), false);
            p.setRemindedBeforeAt(now);
        }
        for (PlanPurchase p : purchases.findByExpiresAtBetweenAndRemindedExpiredAtIsNull(now.minus(Duration.ofDays(1)), now)) {
            if (stillTheEndOf(p)) notifier.planReminderToOwner(p.getTelegramId(), p.getPlanCode(), true);
            p.setRemindedExpiredAt(now);
            p.setRemindedBeforeAt(p.getRemindedBeforeAt() == null ? now : p.getRemindedBeforeAt());
        }
    }

    /** True when this purchase's end is the user's actual end — not overtaken by a renewal or upgrade. */
    private boolean stillTheEndOf(PlanPurchase p) {
        return subscriptions.findById(p.getUserId())
                .map(s -> s.getExpiresAt() != null && s.getExpiresAt().equals(p.getExpiresAt()))
                .orElse(false);
    }
}

package uz.efbazaar.trade.plan.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import uz.efbazaar.trade.mailbox.domain.Mailbox;
import uz.efbazaar.trade.mailbox.repository.MailboxRepository;
import uz.efbazaar.trade.trade.domain.TradeState;
import uz.efbazaar.trade.trade.repository.TradeRepository;

import java.util.*;

/**
 * Which of a user's mailboxes their plan lets them read.
 *
 * <p>Computed on every read, never stored: a plan change, an expiry, an admin
 * cutting the limit, a detach — each shows up on the next request with no job
 * to run and nothing to fall out of step.
 *
 * <p>A mailbox in a live trade is always open and counts toward the limit;
 * locking it mid-trade would strand the trade, so after a downgrade the user
 * may briefly read more than N.
 */
@Service
@RequiredArgsConstructor
public class MailboxAccessService {

    private final MailboxRepository mailboxes;
    private final TradeRepository trades;
    private final EntitlementService entitlements;

    @Transactional(readOnly = true)
    public Set<UUID> lockedIds(UUID userId) {
        OptionalLong limit = entitlements.limit(userId, MailboxSlotService.KEY);
        if (limit.isEmpty()) return Set.of();

        List<Mailbox> ordered = mailboxes.findAccessOrderOwnedBy(userId);
        List<Mailbox> live = new ArrayList<>();
        List<Mailbox> rest = new ArrayList<>();
        for (Mailbox m : ordered) {
            (trades.existsLiveWithMailbox(m.getId(), TradeState.nonTerminal()) ? live : rest).add(m);
        }
        long free = Math.max(0, limit.getAsLong() - live.size());
        Set<UUID> locked = new LinkedHashSet<>();
        for (int i = 0; i < rest.size(); i++) {
            if (i >= free) locked.add(rest.get(i).getId());
        }
        return locked;
    }

    @Transactional(readOnly = true)
    public boolean isLocked(UUID userId, UUID mailboxId) {
        return lockedIds(userId).contains(mailboxId);
    }
}

import { isInsufficientFunds } from './wallet-shortfall';

export type PlanCode = 'STANDARD' | 'PRO' | 'MAX';
export type PlanPeriod = 'MONTH' | 'YEAR';

export interface PlanCatalogItem {
  code: PlanCode;
  sortOrder: number;
  /** null value = unlimited. */
  entitlements: Record<string, string | null>;
  /** Empty for the free plan. Tiyin. */
  prices: Partial<Record<PlanPeriod, number>>;
}

export interface MyPlan {
  code: PlanCode;
  expiresAt: string | null;
  entitlements: Record<string, string | null>;
}

/** Backend codes stay English; these are the only place the names live. */
export const PLAN_NAMES: Record<PlanCode, string> = {
  STANDARD: 'Jaydari',
  PRO: 'Boyvachcha',
  MAX: 'Aristokrat',
};

export const SLOTS_KEY = 'mailbox.slots';
export const CUSTOM_SLOTS_KEY = 'mailbox.custom_slots';

/** null = unlimited; a missing key = the plan does not grant it (0). */
export function slotCount(entitlements: Record<string, string | null>, key: string): number | null {
  if (!(key in entitlements)) return 0;
  const v = entitlements[key];
  if (v === null) return null;
  const n = Number.parseInt(v, 10);
  return Number.isFinite(n) ? n : 0;
}

/** Computed from live prices so an admin edit never leaves a stale badge. */
export function yearlySavingsPercent(prices: Partial<Record<PlanPeriod, number>>): number | null {
  const month = prices.MONTH;
  const year = prices.YEAR;
  if (!month || !year) return null;
  const full = month * 12;
  if (year >= full) return null;
  return Math.floor(((full - year) / full) * 100);
}

function isLive(mine: MyPlan, now: Date): boolean {
  if (mine.code === 'STANDARD') return false;
  return mine.expiresAt === null || new Date(mine.expiresAt).getTime() > now.getTime();
}

export type PlanAction = 'choose' | 'extend' | 'upgrade' | 'unavailable' | 'current-free';

export function planAction(
  target: PlanCatalogItem,
  mine: MyPlan,
  catalog: PlanCatalogItem[],
  now: Date,
): PlanAction {
  const live = isLive(mine, now);
  if (target.code === 'STANDARD') return live ? 'unavailable' : 'current-free';
  if (!live) return 'choose';
  if (target.code === mine.code) return 'extend';
  const currentOrder = catalog.find((p) => p.code === mine.code)?.sortOrder ?? 0;
  return target.sortOrder > currentOrder ? 'upgrade' : 'unavailable';
}

export function daysLeft(expiresAt: string | null, now: Date): number | null {
  if (!expiresAt) return null;
  const ms = new Date(expiresAt).getTime() - now.getTime();
  return ms <= 0 ? 0 : Math.ceil(ms / 86_400_000);
}

export type PurchaseError =
  | { kind: 'insufficient'; message: string }
  | { kind: 'price_changed'; currentPriceTiyin: number }
  | { kind: 'other'; message: string };

export function purchaseErrorFrom(status: number, body: unknown): PurchaseError {
  const b = (body ?? {}) as { code?: string; detail?: string; message?: string; currentPriceTiyin?: number };
  const message = b.detail || b.message || `So'rov bajarilmadi (${status})`;
  if (b.code === 'price_changed' && typeof b.currentPriceTiyin === 'number') {
    return { kind: 'price_changed', currentPriceTiyin: b.currentPriceTiyin };
  }
  if (isInsufficientFunds(message)) return { kind: 'insufficient', message };
  return { kind: 'other', message };
}

export interface PlanPriceRow {
  planCode: PlanCode; period: PlanPeriod; priceTiyin: number; updatedBy: string | null; updatedAt: string;
}
export interface PlanEntitlementRow {
  planCode: PlanCode; key: string; value: string | null; updatedBy: string | null; updatedAt: string;
}
export type SlotDraft = { unlimited: boolean; count: string };

export function toEntitlementValue(d: SlotDraft): string | null {
  if (d.unlimited) return null;
  const s = d.count.trim();
  if (!/^\d+$/.test(s)) throw new Error('Butun son kiriting');
  return String(Number.parseInt(s, 10));
}

export function fromEntitlementValue(v: string | null): SlotDraft {
  return v === null ? { unlimited: true, count: '' } : { unlimited: false, count: v };
}

export function isReduction(before: string | null, after: string | null): boolean {
  if (after === null) return false;
  if (before === null) return true;
  return Number(after) < Number(before);
}

export function parseSomInput(raw: string): number {
  const s = raw.replace(/[\s  ]/g, '');
  if (!/^\d+$/.test(s)) throw new Error('Narxni so‘mda butun son bilan kiriting');
  const som = Number.parseInt(s, 10);
  if (som <= 0) throw new Error('Narx 0 dan katta bo‘lishi kerak');
  return som * 100;
}

const BASE = '/api/payment/v1/admin/plans';

async function call<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    ...init,
    headers: { 'Content-Type': 'application/json', Accept: 'application/json', ...(init?.headers ?? {}) },
  });
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body.detail || body.message || `So'rov bajarilmadi (${res.status})`);
  }
  return (await res.json()) as T;
}
