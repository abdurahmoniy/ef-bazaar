# Marketplace Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first auth-aware marketplace frontend slice: public browse/detail, protected listing actions, a minimal create listing flow, and landing-page integration.

**Architecture:** Add focused marketplace domain files under `frontend/src/lib`, reusable marketplace UI under `frontend/src/components/marketplace`, and route-level pages/actions under `frontend/src/app/marketplace`. Public pages call marketplace APIs with optional auth cookies; protected actions stay in server actions so access tokens remain httpOnly.

**Tech Stack:** Next.js 16 App Router, React 19 server actions, TypeScript strict, Tailwind 3.4, shadcn-style local components, Vitest for pure helper tests.

```ts
import type { ListingBrowseParams } from './marketplace-types';

export type SearchParamInput = Record<string, string | string[] | undefined>;

const UZ_MONTHS = [
  'yanvar',
  'fevral',
  'mart',
  'aprel',
  'may',
  'iyun',
  'iyul',
  'avgust',
  'sentabr',
  'oktabr',
  'noyabr',
  'dekabr',
];

export const MARKETPLACE_FILTERS = [
  { id: 'all', label: 'Barchasi', params: {} },
  { id: 'negotiable', label: 'Savdolashish', params: { priceNegotiable: true } },
  { id: 'google', label: 'Google linked', params: { hasGooglePlayLink: true } },
  { id: 'division-1-3', label: 'Div 1-3', params: { minDivision: 1, maxDivision: 3 } },
  { id: 'strong', label: '3000+ strength', params: { minTeamStrength: 3000 } },
] as const satisfies readonly {
  id: string;
  label: string;
  params: ListingBrowseParams;
}[];

export type MarketplaceFilterId = (typeof MARKETPLACE_FILTERS)[number]['id'];

const FILTER_PARAMS = MARKETPLACE_FILTERS.reduce(
  (acc, filter) => {
    acc[filter.id] = filter.params;
    return acc;
  },
  {} as Record<MarketplaceFilterId, ListingBrowseParams>,
);

function isMarketplaceFilterId(value: string): value is MarketplaceFilterId {
  return value in FILTER_PARAMS;
}

function firstParam(value: string | string[] | undefined) {
  return Array.isArray(value) ? value[0] : value;
}

function textParam(value: string | string[] | undefined) {
  const text = firstParam(value)?.trim();
  return text ? text : undefined;
}

function numberParam(value: string | string[] | undefined, min = 0) {
  const text = textParam(value);
  if (!text) return undefined;

  const parsed = Number(text);
  if (!Number.isFinite(parsed) || parsed < min) return undefined;

  return Math.trunc(parsed);
}

function booleanParam(value: string | string[] | undefined) {
  const text = textParam(value);
  if (text === 'true') return true;
  if (text === 'false') return false;
  return undefined;
}

function formatInteger(value: number) {
  return String(Math.round(value)).replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
}

function appendDefined(params: URLSearchParams, key: string, value: unknown) {
  if (value === undefined || value === null || value === '') return;
  if (typeof value === 'string') {
    const trimmed = value.trim();
    if (trimmed) params.set(key, trimmed);
    return;
  }
  params.set(key, String(value));
}

export function formatPriceUZS(value: number | string) {
  const numeric = Number(value);
  if (!Number.isFinite(numeric)) return "0 so'm";
  return `${formatInteger(numeric)} so'm`;
}

export function formatCompactNumber(value: number | string) {
  const numeric = Number(value);
  if (!Number.isFinite(numeric)) return '0';
  return formatInteger(numeric);
}

export function formatListingDate(value: string | Date) {
  const date = typeof value === 'string' ? new Date(value) : value;
  if (Number.isNaN(date.getTime())) return '';

  const uzDate = new Date(date.getTime() + 5 * 60 * 60 * 1000);
  const day = String(uzDate.getUTCDate()).padStart(2, '0');
  const month = UZ_MONTHS[uzDate.getUTCMonth()];
  const hour = String(uzDate.getUTCHours()).padStart(2, '0');
  const minute = String(uzDate.getUTCMinutes()).padStart(2, '0');

  return `${day} ${month} ${hour}:${minute}`;
}

export function divisionLabel(value: number) {
  if (!Number.isInteger(value) || value < 1 || value > 9) return 'Division -';
  return `Division ${value}`;
}

export function normalizePlayerList(value: string) {
  return value
    .split(/[,\n]/)
    .map((item) => item.trim())
    .filter(Boolean)
    .slice(0, 12);
}

export function getSelectedMarketplaceFilter(input: SearchParamInput = {}): MarketplaceFilterId {
  const filter = textParam(input.filter);
  return filter && isMarketplaceFilterId(filter) ? filter : 'all';
}

export function parseListingSearchParams(input: SearchParamInput = {}): ListingBrowseParams {
  const selectedFilter = getSelectedMarketplaceFilter(input);
  const params: ListingBrowseParams = { ...FILTER_PARAMS[selectedFilter] };

  const page = numberParam(input.page, 0);
  const size = numberParam(input.size, 1);
  const minPrice = numberParam(input.minPrice, 0);
  const maxPrice = numberParam(input.maxPrice, 0);
  const minDivision = numberParam(input.minDivision, 1);
  const maxDivision = numberParam(input.maxDivision, 1);
  const minTeamStrength = numberParam(input.minTeamStrength, 0);
  const hasGooglePlayLink = booleanParam(input.hasGooglePlayLink);
  const priceNegotiable = booleanParam(input.priceNegotiable);
  const search = textParam(input.q) ?? textParam(input.search);
  const valuablePlayer = textParam(input.valuablePlayer);

  if (page !== undefined) params.page = page;
  if (size !== undefined) params.size = size;
  if (minPrice !== undefined) params.minPrice = minPrice;
  if (maxPrice !== undefined) params.maxPrice = maxPrice;
  if (minDivision !== undefined) params.minDivision = minDivision;
  if (maxDivision !== undefined) params.maxDivision = maxDivision;
  if (minTeamStrength !== undefined) params.minTeamStrength = minTeamStrength;
  if (hasGooglePlayLink !== undefined) params.hasGooglePlayLink = hasGooglePlayLink;
  if (priceNegotiable !== undefined) params.priceNegotiable = priceNegotiable;
  if (valuablePlayer) params.valuablePlayer = valuablePlayer;
  if (search) params.search = search;

  return params;
}

export function buildListingSearchParams(input: ListingBrowseParams = {}) {
  const params = new URLSearchParams();

  appendDefined(params, 'page', input.page);
  appendDefined(params, 'size', input.size);
  appendDefined(params, 'sort', input.sort ?? 'createdAt,desc');
  appendDefined(params, 'minPrice', input.minPrice);
  appendDefined(params, 'maxPrice', input.maxPrice);
  appendDefined(params, 'minDivision', input.minDivision);
  appendDefined(params, 'maxDivision', input.maxDivision);
  appendDefined(params, 'minTeamStrength', input.minTeamStrength);
  appendDefined(params, 'hasGooglePlayLink', input.hasGooglePlayLink);
  appendDefined(params, 'priceNegotiable', input.priceNegotiable);
  appendDefined(params, 'valuablePlayer', input.valuablePlayer);
  appendDefined(params, 'search', input.search);

  return params;
}

import 'server-only';

import { apiFetch } from './api';
import { buildListingSearchParams } from './marketplace-format';
import type {
  CartItemResponse,
  ChatResponse,
  LikeToggleResponse,
  ListingBrowseParams,
  ListingCreateRequest,
  ListingResponse,
  PageResponse,
} from './marketplace-types';

const MARKETPLACE_ROOT = '/marketplace/v1';

function withQuery(path: string, params?: ListingBrowseParams) {
  const query = buildListingSearchParams(params);
  const serialized = query.toString();
  return serialized ? `${path}?${serialized}` : path;
}

export async function browseListings(params: ListingBrowseParams = {}) {
  return apiFetch<PageResponse<ListingResponse>>(
    withQuery(`${MARKETPLACE_ROOT}/listings`, params),
  );
}

export async function getListing(id: string) {
  return apiFetch<ListingResponse>(`${MARKETPLACE_ROOT}/listings/${id}`);
}

export async function getMyListings(params: ListingBrowseParams = {}) {
  return apiFetch<PageResponse<ListingResponse>>(
    withQuery(`${MARKETPLACE_ROOT}/listings/mine`, params),
  );
}

export async function createListing(request: ListingCreateRequest) {
  return apiFetch<ListingResponse>(`${MARKETPLACE_ROOT}/listings`, {
    method: 'POST',
    body: request,
  });
}

export async function toggleListingLike(id: string) {
  return apiFetch<LikeToggleResponse>(`${MARKETPLACE_ROOT}/listings/${id}/like`, {
    method: 'POST',
  });
}

export async function listCart() {
  return apiFetch<CartItemResponse[]>(`${MARKETPLACE_ROOT}/cart`);
}

export async function addToCart(listingId: string) {
  return apiFetch<CartItemResponse>(`${MARKETPLACE_ROOT}/cart/${listingId}`, {
    method: 'POST',
  });
}

export async function removeFromCart(listingId: string) {
  return apiFetch<void>(`${MARKETPLACE_ROOT}/cart/${listingId}`, {
    method: 'DELETE',
  });
}

export async function startChat(listingId: string) {
  return apiFetch<ChatResponse>(`${MARKETPLACE_ROOT}/chats`, {
    method: 'POST',
    body: { listingId },
  });
}

'use server';

import { revalidatePath } from 'next/cache';
import { redirect } from 'next/navigation';

import { ApiException } from '@/lib/api';
import {
  addToCart,
  createListing,
  removeFromCart,
  startChat,
  toggleListingLike,
} from '@/lib/marketplace';
import { normalizePlayerList } from '@/lib/marketplace-format';
import type { ListingCreateRequest } from '@/lib/marketplace-types';

export interface MarketplaceActionResult {
  ok: boolean;
  error?: string;
  code?: string;
  liked?: boolean;
  totalLikes?: number;
  chatId?: string;
}

export interface ListingFormState extends MarketplaceActionResult {
  listingId?: string;
}

const LOGIN_REQUIRED_MESSAGE = 'Telegram orqali kirish kerak';

function toActionError(error: unknown, fallback: string): MarketplaceActionResult {
  if (error instanceof ApiException) {
    if (error.status === 401 || error.status === 403) {
      return {
        ok: false,
        error: LOGIN_REQUIRED_MESSAGE,
        code: error.code ?? 'UNAUTHORIZED',
      };
    }
    return { ok: false, error: error.message, code: error.code };
  }
  return { ok: false, error: fallback };
}

function textField(formData: FormData, key: string) {
  const value = formData.get(key);
  return typeof value === 'string' ? value.trim() : '';
}

function numberField(formData: FormData, key: string) {
  const value = Number(textField(formData, key));
  return Number.isFinite(value) ? value : Number.NaN;
}

function createRequestFromForm(formData: FormData): ListingCreateRequest | string {
  const description = textField(formData, 'description');
  const highestDivisionLevel = numberField(formData, 'highestDivisionLevel');
  const androidCoins = numberField(formData, 'androidCoins');
  const iosCoins = numberField(formData, 'iosCoins');
  const teamStrength = numberField(formData, 'teamStrength');
  const price = numberField(formData, 'price');

  if (description.length < 10) return "Tavsif kamida 10 ta belgidan iborat bo'lishi kerak";
  if (!Number.isInteger(highestDivisionLevel) || highestDivisionLevel < 1 || highestDivisionLevel > 9) {
    return "Division 1 dan 9 gacha bo'lishi kerak";
  }
  if (androidCoins < 0 || iosCoins < 0 || teamStrength < 0 || price < 0) {
    return "Raqamli maydonlar manfiy bo'lmasligi kerak";
  }

  return {
    description,
    valuablePlayers: normalizePlayerList(textField(formData, 'valuablePlayers')),
    hasGooglePlayLink: formData.get('hasGooglePlayLink') === 'on',
    highestDivisionLevel,
    androidCoins,
    iosCoins,
    teamStrength,
    price,
    priceNegotiable: formData.get('priceNegotiable') === 'on',
  };
}

export async function toggleListingLikeAction(listingId: string): Promise<MarketplaceActionResult> {
  try {
    const result = await toggleListingLike(listingId);
    revalidatePath('/marketplace');
    revalidatePath(`/marketplace/${listingId}`);
    return {
      ok: true,
      liked: result.liked,
      totalLikes: result.totalLikes,
    };
  } catch (error) {
    return toActionError(error, 'Like saqlanmadi');
  }
}

export async function addListingToCartAction(listingId: string): Promise<MarketplaceActionResult> {
  try {
    await addToCart(listingId);
    revalidatePath('/marketplace');
    revalidatePath(`/marketplace/${listingId}`);
    return { ok: true };
  } catch (error) {
    return toActionError(error, "Savatga qo'shilmadi");
  }
}

export async function removeListingFromCartAction(listingId: string): Promise<MarketplaceActionResult> {
  try {
    await removeFromCart(listingId);
    revalidatePath('/marketplace');
    revalidatePath(`/marketplace/${listingId}`);
    return { ok: true };
  } catch (error) {
    return toActionError(error, 'Savatdan olib tashlanmadi');
  }
}

export async function startListingChatAction(listingId: string): Promise<MarketplaceActionResult> {
  try {
    const chat = await startChat(listingId);
    return { ok: true, chatId: chat.id };
  } catch (error) {
    return toActionError(error, 'Chat ochilmadi');
  }
}

export async function createListingAction(
  _previousState: ListingFormState,
  formData: FormData,
): Promise<ListingFormState> {
  const request = createRequestFromForm(formData);
  if (typeof request === 'string') return { ok: false, error: request };

  let listingId: string;
  try {
    const listing = await createListing(request);
    listingId = listing.id;
  } catch (error) {
    return toActionError(error, "E'lon yaratilmadi");
  }

  revalidatePath('/marketplace');
  redirect(`/marketplace/${listingId}`);
}
```

function filterHref(filterId: MarketplaceFilterId, query?: string) {
  const params = new URLSearchParams();
